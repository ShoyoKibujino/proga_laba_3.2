    def foo(n):

    """!!!
    Алгоритм: Функция возвращает список всех простых чисел от 1 до n включительно
    Сложность: O(N^2), так как внешний цикл выполняется N раз (от 1 до n), внутренний цикл (while): для каждого i выполняется до i итераций
    Вычислительная сложность для этого алгоритма ХУДШАЯ, так как для каждого числа i от 1 до n программа проверяет ВСЕ числа от 2 до i-1.
    
    параметры:
    - n (int): число.
    
    Сложность: O(N^2).
    """
    
    res = []
    for i in range(1, n + 1):
        divisors = 0
        j = 2
        while j < i and divisors == 0:
            if i % j == 0:
                divisors += 1
            j += 1
        
        if divisors == 0:
            res.append(i)
    
    return res

Для этой задачи алгоритмом с ЛУЧШЕЙ вычислительной сложностью будет алгоритм Решето Эратосфена

    def foo(n):
  
    """
    Алгоритм: Решето Эратосфена.
    Сложность: O(N log log N) так как не проверяем каждое число, а вычеркиваем кратные простым
    """
    if n < 2:
        return []
    
    is_prime = [True] * (n + 1)
    is_prime[0] = is_prime[1] = False
    
    for i in range(2, int(n ** 0.5) + 1):
        if is_prime[i]:
            for j in range(i * i, n + 1, i):
                is_prime[j] = False
    
    return [i for i in range(2, n + 1) if is_prime[i]]

O(N log log N) — оптимальный для нахождения всех простых чисел до n. Использует дополнительную память O(N), но выигрывает в скорости
